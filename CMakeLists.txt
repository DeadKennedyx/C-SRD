cmake_minimum_required(VERSION 3.20)
project(csrd VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CSRD_BUILD_FRONTEND "Build the Clang LibTooling transformer" ON)
include(CTest)

add_library(csrd_representation src/representation/representation.cpp src/representation/costs.cpp src/codegen/graph_emitter.cpp src/analysis/graph.cpp src/transforms/algorithms.cpp)
target_include_directories(csrd_representation PUBLIC include)
target_compile_options(csrd_representation PRIVATE -Wall -Wextra -Wpedantic)

if(CSRD_BUILD_FRONTEND)
  find_package(LLVM REQUIRED CONFIG)
  find_package(Clang REQUIRED CONFIG)
  message(STATUS "Using LLVM ${LLVM_PACKAGE_VERSION} from ${LLVM_DIR}")
  find_program(CSRD_CLANG_EXECUTABLE NAMES clang HINTS "${LLVM_TOOLS_BINARY_DIR}" NO_DEFAULT_PATH REQUIRED)
  execute_process(COMMAND "${CSRD_CLANG_EXECUTABLE}" -print-resource-dir
    OUTPUT_VARIABLE CSRD_RESOURCE_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(APPLE)
    execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE CSRD_SYSROOT
      OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  endif()
  configure_file(include/csrd/config.hpp.in generated/csrd/config.hpp @ONLY)
  find_path(Z3_INCLUDE_DIR z3++.h HINTS /opt/homebrew/include REQUIRED)
  find_library(Z3_LIBRARY z3 HINTS /opt/homebrew/lib REQUIRED)
  add_executable(csrd src/main.cpp src/frontend/frontend.cpp src/frontend/metrics.cpp src/analysis/analyzer.cpp src/verification/translation.cpp src/transforms/planner.cpp)
  target_include_directories(csrd PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_include_directories(csrd SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
  target_include_directories(csrd SYSTEM PRIVATE ${Z3_INCLUDE_DIR})
  target_compile_options(csrd PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(csrd PRIVATE csrd_representation clang-cpp LLVM ${Z3_LIBRARY})
endif()

if(BUILD_TESTING)
  add_executable(representation_test tests/unit/representation_test.cpp)
  target_link_libraries(representation_test PRIVATE csrd_representation)
  add_test(NAME representation COMMAND representation_test)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  if(CSRD_BUILD_FRONTEND)
    add_test(NAME integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_cli.py" $<TARGET_FILE:csrd>)
    add_test(NAME translation_validation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_verification.py" $<TARGET_FILE:csrd>)
    add_test(NAME representation_planning COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_planning.py" $<TARGET_FILE:csrd>)
    add_test(NAME algorithm_candidates COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_algorithms.py" $<TARGET_FILE:csrd>)
    add_test(NAME defensive_evaluation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_evaluation.py" $<TARGET_FILE:csrd>)
    add_test(NAME differential COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/differential/run.py" --csrd $<TARGET_FILE:csrd> --report "${CMAKE_CURRENT_BINARY_DIR}/differential.json")
    add_test(NAME expanded_differential COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/differential/expanded.py" --csrd $<TARGET_FILE:csrd> --report "${CMAKE_CURRENT_BINARY_DIR}/expanded-differential.json")
    set_tests_properties(expanded_differential PROPERTIES TIMEOUT 600)
    set_tests_properties(differential PROPERTIES TIMEOUT 240)
  endif()
  find_program(Z3_EXECUTABLE z3)
  if(Z3_EXECUTABLE)
    add_test(NAME rule_contracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/verify_rules.py" --z3 "${Z3_EXECUTABLE}" --report "${CMAKE_CURRENT_BINARY_DIR}/rule-contracts.json")
    set_tests_properties(rule_contracts PROPERTIES TIMEOUT 90)
  endif()
endif()
