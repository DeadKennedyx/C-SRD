  find_package(LLVM REQUIRED CONFIG)
  find_package(Clang REQUIRED CONFIG)
  message(STATUS "Using LLVM ${LLVM_PACKAGE_VERSION} from ${LLVM_DIR}")
  find_program(CSRD_CLANG_EXECUTABLE NAMES clang HINTS "${LLVM_TOOLS_BINARY_DIR}" NO_DEFAULT_PATH REQUIRED)
  execute_process(COMMAND "${CSRD_CLANG_EXECUTABLE}" -print-resource-dir
    OUTPUT_VARIABLE CSRD_RESOURCE_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(APPLE)
    execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE CSRD_SYSROOT
      OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  endif()

  find_path(Z3_INCLUDE_DIR z3++.h HINTS /opt/homebrew/include REQUIRED)
  find_library(Z3_LIBRARY z3 HINTS /opt/homebrew/lib REQUIRED)

  target_include_directories(csrd_verification SYSTEM PRIVATE ${Z3_INCLUDE_DIR})
  target_compile_options(csrd_verification PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(csrd_verification PUBLIC csrd_representation PRIVATE ${Z3_LIBRARY})
  add_library(csrd_private_analysis src/analysis/private_regions.cpp)
  target_include_directories(csrd_private_analysis SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
  target_compile_options(csrd_private_analysis PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(csrd_private_analysis PUBLIC csrd_verification PRIVATE clang-cpp LLVM)
  add_library(csrd_control src/analysis/control_regions.cpp src/codegen/control_emitter.cpp src/verification/control.cpp src/analysis/outline.cpp)
  target_include_directories(csrd_control PUBLIC include)
  target_include_directories(csrd_control SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
  target_compile_options(csrd_control PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(csrd_control PRIVATE clang-cpp LLVM)
  add_executable(csrd src/main.cpp src/frontend/frontend.cpp src/frontend/compilation.cpp src/frontend/contracts.cpp src/frontend/metrics.cpp src/analysis/analyzer.cpp src/transforms/planner.cpp)
  target_include_directories(csrd PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_include_directories(csrd SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
  target_include_directories(csrd SYSTEM PRIVATE ${Z3_INCLUDE_DIR})
  target_compile_options(csrd PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(csrd PRIVATE csrd_private_analysis csrd_control clang-cpp LLVM ${Z3_LIBRARY})
  add_executable(csrd-query-circuit tools/query_circuit.cpp)
  target_include_directories(csrd-query-circuit SYSTEM PRIVATE ${Z3_INCLUDE_DIR})
  target_compile_options(csrd-query-circuit PRIVATE -Wall -Wextra -Wpedantic)
